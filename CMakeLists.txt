cmake_minimum_required(VERSION 3.24)
project(Venor VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(VENOR_APP_NAME "Venor" CACHE STRING "Configurable working application title")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Venor requires Windows x64")
endif()
enable_language(RC)
set(VENOR_ICON_FILE "${CMAKE_CURRENT_SOURCE_DIR}/venor.ico")
configure_file(resources/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Concurrent Sql Test)
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/zydis EXCLUDE_FROM_ALL)
add_library(venor_core src/analysis.cpp src/semantics.cpp src/workspace.cpp src/debugger.cpp src/catalog.cpp)
target_include_directories(venor_core PUBLIC src)
target_compile_definitions(venor_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN APP_TITLE="${VENOR_APP_NAME}")
target_link_libraries(venor_core PUBLIC Qt6::Widgets Qt6::Concurrent Qt6::Sql Zydis ole32 dbgeng)
add_library(venor_ui src/mainwindow.cpp src/mainwindow.h)
target_link_libraries(venor_ui PUBLIC venor_core)
add_executable(venor WIN32 src/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
qt_add_resources(venor "app_assets"
  PREFIX "/icons"
  BASE "${CMAKE_CURRENT_SOURCE_DIR}"
  FILES "${VENOR_ICON_FILE}"
)
target_link_libraries(venor PRIVATE venor_ui)
add_executable(benign_target samples/benign_target.cpp)
add_library(benign_library SHARED samples/benign_library.cpp)
target_compile_options(benign_target PRIVATE -O0 -g)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE venor_core Qt6::Test)
add_test(NAME core COMMAND core_tests $<TARGET_FILE:benign_target> $<TARGET_FILE:benign_library>)
add_executable(debugger_tests tests/debugger_tests.cpp)
target_link_libraries(debugger_tests PRIVATE venor_core Qt6::Test)
add_test(NAME debugger COMMAND debugger_tests $<TARGET_FILE:benign_target>)
set_tests_properties(debugger PROPERTIES TIMEOUT 90)
add_executable(gui_tests tests/gui_tests.cpp)
target_link_libraries(gui_tests PRIVATE venor_ui Qt6::Test)
add_test(NAME gui COMMAND gui_tests $<TARGET_FILE:benign_target>)
set_tests_properties(gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
install(TARGETS venor benign_target RUNTIME DESTINATION .)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
